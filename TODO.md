## Архитектура данных

### Вариант упражнения (расширенный)
```javascript
{
  id: 'var-123456',
  baseExerciseId: 'base-789',
  name: 'Жим штанги лежа',          // отображаемое название
  normalizedName: 'жим штанги лежа', // для сравнения
  originalName: 'Жим штанги лежа',   // оригинал из парсера
  originalText: '1) Жим штанги лежа', // полный оригинальный текст строки
  type: 'strength',                   // strength | cardio | bodyweight | timed
  metricType: 'weight',               // weight | reps | time | cardio | time_weight | distance
  needsReview: false,                 // true если формат не распознан
  categoryId: 'cat-strength'          // категория (дублируется из baseExercise для быстрого доступа)
}
```

### Прогресс (расширенный)
```javascript
{
  "var-123456": [
    {
      date: "2026-07-24",
      workoutId: "w-123",
      sets: 4,
      exerciseId: "var-123456",
      baseExerciseId: "base-789",
      type: "strength",
      metricType: "weight",
      maxWeight: 80,
      totalReps: 40,
      volume: 3200,
      originalText: "20кг*10\n20кг*10\n20кг*10\n20кг*10" // оригинальные строки
    }
  ]
}
```
